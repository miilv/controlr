import fcntl, hashlib, json, os, subprocess, sys, time
from pathlib import Path
H=Path.home(); ROOT=H/'dojo-data/kong-retest-eligible-1107'; REPO=Path(sys.argv[1]).resolve()
sys.path.insert(0,str(REPO))
from controlr.keyguard import child_env
D=json.loads((ROOT/'design.json').read_text())

def note(msg):
    with (ROOT/'notes.log').open('a') as f:f.write(time.strftime('%Y-%m-%dT%H:%M:%SZ ',time.gmtime())+msg+'\n')

def run(tag, args, env, gpu=False):
    done=ROOT/(tag+'.done')
    if done.exists():return
    lock=None
    if gpu:
        deadline=time.monotonic()+9000
        while time.monotonic()<deadline:
            used=int(subprocess.check_output(['nvidia-smi','--query-gpu=memory.used','--format=csv,noheader,nounits'],text=True).strip())
            load=os.getloadavg()[0]
            if used<=20480 and load<=26:
                for name in ['dojo-gpu-light.lock']+[f'dojo-gpu-light{i}.lock' for i in range(1,9)]:
                    p=H/name
                    if not p.exists() or p.is_symlink():continue
                    f=p.open('a+')
                    try:fcntl.flock(f,fcntl.LOCK_EX|fcntl.LOCK_NB)
                    except BlockingIOError:f.close();continue
                    lock=f;break
            if lock:break
            time.sleep(20)
        if not lock:raise RuntimeError('No admitted free light lock before prep deadline')
        used=int(subprocess.check_output(['nvidia-smi','--query-gpu=memory.used','--format=csv,noheader,nounits'],text=True).strip())
        if used>20480 or os.getloadavg()[0]>26:lock.close();return run(tag,args,env,gpu)
        lock.seek(0);lock.truncate();lock.write(f'pid={os.getpid()} stream=KONG prep={tag}\n');lock.flush()
        note(f'PREP START {tag} lock={lock.name} used_MiB={used} load={os.getloadavg()[0]}')
    cmd=['bash',str(REPO/'scripts/simlab/rdpy.sh')]+args
    if gpu:cmd=['systemd-run','--user','--scope','-q',f'--unit=kong_rt_prep_{tag}_{os.getpid()}','-p','MemoryMax=20G','-p','CPUQuota=600%']+cmd
    try:
        with (ROOT/(tag+'.log')).open('a') as out:
            rc=subprocess.run(cmd,cwd=REPO,env=env,stdout=out,stderr=subprocess.STDOUT).returncode
        note(f'PREP END {tag} rc={rc}')
        if rc:raise RuntimeError(f'{tag}: rc={rc}, see log')
        done.touch()
    finally:
        if lock:lock.close()
